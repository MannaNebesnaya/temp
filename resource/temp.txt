    private List<PaymentMethodData> buildPaymentMethodData(final QrPutRequest req,
                                                           final QrData qrData,
                                                           final QrEntityGetPaymentServicesResponse response) {
        final boolean accountSpecified = req.getAccount() != null || req.getAccAlias() != null;
 
        // Ветка 1: блок paymentMethodData НЕ пришёл в запросе
        if (req.getPaymentMethodData() == null || req.getPaymentMethodData().isEmpty()) {
            return buildWhenRequestHasNoBlock(qrData, response, accountSpecified);
        }
        // Ветка 2: блок пришёл -> формируем по ответу sbp-qr-entity (code == 0)
        return buildFromResponse(response);
    }
 
    // Ветка 1
    private List<PaymentMethodData> buildWhenRequestHasNoBlock(final QrData qrData,
                                                               final QrEntityGetPaymentServicesResponse response,
                                                               final boolean accountSpecified) {
        final List<PaymentMethodData> stored = qrData.getPaymentMethodData();
 
        // 1a: qr_data.payment_method_data не пусто -> берём как есть (копия, чтобы не мутировать jsonb сущности)
        if (stored != null && !stored.isEmpty()) {
            final List<PaymentMethodData> result = stored.stream().map(this::copy).collect(Collectors.toList());
            if (accountSpecified) {
                // в объектах PS0000000001 / PS0000000006 обновить receiverAccount из ответа (по совпадающему id)
                result.stream()
                        .filter(b -> PS_SBP.equals(b.getPaymentServiceId()) || PS_MIRLINK.equals(b.getPaymentServiceId()))
                        .forEach(b -> additionalData(b).setReceiverAccount(
                                accountFromResponse(response, b.getPaymentServiceId())));
            }
            return result;
        }
 
        // 1b: qr_data.payment_method_data пусто -> только блок СБП
        final AdditionalData additional = new AdditionalData();
        additional.setMemberId(resolveMemberId());
        if (accountSpecified) {
            // доке: account из ответа, где paymentServiceId IN (PS0000000001, PS0000000006).
            // Для одиночного СБП-блока берём СБП, при отсутствии — МирЛинк. (Уточнить трактовку "IN".)
            String account = accountFromResponse(response, PS_SBP);
            if (account == null) {
                account = accountFromResponse(response, PS_MIRLINK);
            }
            additional.setReceiverAccount(account);
        } else {
            additional.setReceiverAccount(qrData.getAccount());
        }
 
        final List<PaymentMethodData> result = new ArrayList<>();
        result.add(new PaymentMethodData(PS_SBP, VERSION, additional));
        return result;
    }
 
    // Ветка 2: каждый блок формируется, только если в ответе есть его paymentServiceId с code == 0.
    // (По видимой доке набор блоков НЕ фильтруется содержимым запроса — только триггерится им. Уточнить.)
    private List<PaymentMethodData> buildFromResponse(final QrEntityGetPaymentServicesResponse response) {
        final List<PaymentMethodData> result = new ArrayList<>();
 
        if (existsWithCodeZero(response, PS_SBP)) {
            final AdditionalData a = new AdditionalData();
            a.setMemberId(resolveMemberId());
            a.setReceiverAccount(accountFromResponse(response, PS_SBP));
            // fraudScore — не формировать
            result.add(new PaymentMethodData(PS_SBP, VERSION, a));
        }
        if (existsWithCodeZero(response, PS_CR)) {
            final AdditionalData a = new AdditionalData();
            a.setReceiverAccount(accountFromResponse(response, PS_CR));
            result.add(new PaymentMethodData(PS_CR, VERSION, a));
        }
        if (existsWithCodeZero(response, PS_MIRLINK)) {
            final AdditionalData a = new AdditionalData();
            a.setReceiverAccount(accountFromResponse(response, PS_MIRLINK));
            result.add(new PaymentMethodData(PS_MIRLINK, VERSION, a));
        }
        return result;
    }
 
    // ---- helpers ---------------------------------------------------------
 
    private Optional<PaymentMethodDataGetPSResponse> find(final QrEntityGetPaymentServicesResponse response,
                                                          final String paymentServiceId) {
        if (response.getPaymentMethodData() == null) {
            return Optional.empty();
        }
        return response.getPaymentMethodData().stream()
                .filter(m -> paymentServiceId.equals(m.getPaymentServiceId()))
                .findFirst();
    }
 
    private boolean existsWithCodeZero(final QrEntityGetPaymentServicesResponse response, final String paymentServiceId) {
        return find(response, paymentServiceId)
                .map(PaymentMethodDataGetPSResponse::getCode)
                .filter(code -> code == 0)
                .isPresent();
    }
 
    private String accountFromResponse(final QrEntityGetPaymentServicesResponse response, final String paymentServiceId) {
        return find(response, paymentServiceId)
                .map(PaymentMethodDataGetPSResponse::getAdditionalData)
                .map(AdditionalDataGetPSResponse::getAccount)
                .orElse(null);
    }
 
    private AdditionalData additionalData(final PaymentMethodData block) {
        if (block.getAdditionalData() == null) {
            block.setAdditionalData(new AdditionalData());
        }
        return block.getAdditionalData();
    }
 
    private PaymentMethodData copy(final PaymentMethodData src) {
        final AdditionalData s = src.getAdditionalData();
        final AdditionalData add = (s == null) ? null : new AdditionalData(
                s.getMemberId(), s.getReceiverAccount(), s.getFraudScore(), s.getOpkcFraudScore(), s.getGostQrData());
        return new PaymentMethodData(src.getPaymentServiceId(), src.getVersion(), add);
    }
