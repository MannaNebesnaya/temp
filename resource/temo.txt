 
@Component
public class QrPutResponseMapper {
 
    public QrPutResponse toResponse(final QrData qrData, final String qrcId) {
        final QrPutResponse response = new QrPutResponse();
        response.setQrcId(qrcId);                              // из параметров запроса
        response.setLocalQrcId(qrData.getLocalQrcId());
        response.setAccAlias(qrData.getAccAlias());
        response.setMerchantId(qrData.getMerchantId());
        response.setAmount(toInteger(qrData.getAmount()));
        response.setAmountEditable(qrData.getAmountEditable());
        response.setPaymentPurpose(qrData.getPaymentPurpose());
        response.setPayload(qrData.getPayload());
        response.setRedirectUrl(qrData.getRedirectUrl());
        response.setPaymentMethodData(paymentMethodData(qrData));
        return response;
    }
 
    // В ответе — только список paymentServiceId из qr_data.payment_method_data.
    private List<PaymentMethodDataElementRequest> paymentMethodData(final QrData qrData) {
        if (qrData.getPaymentMethodData() == null) {
            return null;
        }
        return qrData.getPaymentMethodData().stream()
                .map(PaymentMethodData::getPaymentServiceId)
                .map(PaymentMethodDataElementRequest::new)
                .collect(Collectors.toList());
    }
 
    // ВНИМАНИЕ: QrPutResponse.amount — Integer, а qr_data.amount — Long.
    // Сумма в копейках (до 12 знаков) переполняет Integer (> ~21,5 млн руб). Скорее всего, в QrPutResponse поле должно быть Long.
    private Integer toInteger(final Long amount) {
        return amount == null ? null : amount.intValue();
    }
}
